## 1. Strategiskais princips: Peļņas virzīts SEO (Profit-First SEO)
SEO nav tikai bezmaksas apmeklējumi — tas ir augstākās peļņas maržas (CM2) virzītājspēks. Kamēr PPC maksā par katru klikšķi, optimizēti saraksti (listings) nodrošina augstu dabīgo rangu, uzlabo reklāmu kvalitātes rādītājus (Quality Score / Ad Rank) un samazina Blended CAC.

                        <span className="text-[10px] font-mono px-1.5 py-0.5 rounded bg-slate-200 text-slate-700">
                          {row.currency}
                        </span>
                      )}
                    </div>
                  </td>
                  <td className="py-3 px-4">
                    <span
                      className={`inline-flex px-2 py-0.5 rounded text-xs font-semibold ${
                        row.status === 'active'
                          ? 'bg-emerald-100 text-emerald-800'
                          : row.status === 'integrating'
                          ? 'bg-amber-100 text-amber-800'
                          : 'bg-slate-100 text-slate-600'
                      }`}
                    >
                      {row.status === 'active' ? 'Aktīvs' : row.status === 'integrating' ? 'Tiek pieslēgts' : 'Plānots'}
                    </span>
                  </td>
                  <td className="py-3 px-4 text-right font-semibold text-slate-900">
                    {formatEur(row.revenueEur)}
                    {row.currency !== 'EUR' && (
                      <div className="text-[11px] text-slate-400 font-normal">
                        ({row.nativeRevenue.toLocaleString()} {row.currency})
                      </div>
                    )}
                  </td>
                  <td className="py-3 px-4 text-center">
                    {row.status === 'active' ? formatPercent(row.revenueTrendPct) : <span className="text-slate-400">-</span>}
                  </td>
                  <td className="py-3 px-4 text-right text-slate-700">
                    {row.ordersCount}
                  </td>
                  <td className="py-3 px-4 text-right text-slate-600">
                    {formatEur(row.adSpendEur)}
                  </td>
                  <td className="py-3 px-4 text-right font-semibold text-emerald-700">
                    {formatEur(row.cm2ProfitEur)}
                  </td>
                  <td className="py-3 px-4 text-right font-semibold">
                    <span className={row.cm2MarginPct >= 30 ? 'text-emerald-700' : 'text-amber-700'}>
                      {row.cm2MarginPct.toFixed(1)}%
                    </span>
                  </td>
                  <td className="py-3 px-4 text-center">
                    {row.isEstimated ? (
                      <span className="text-[11px] bg-amber-50 text-amber-700 border border-amber-200 px-1.5 py-0.5 rounded" title="Aprēķināts no daļējiem datiem">
                        Aprēķināts
                      </span>
                    ) : (
                      <span className="text-[11px] bg-slate-100 text-slate-600 px-1.5 py-0.5 rounded" title={`Sinhronizēts: ${row.lastSyncedAt}`}>
                        Tiešraide
                      </span>
                    )}
                  </td>
                </tr>
              ))}
            </tbody>
          </table>
        </div>
      </div>
    </div>
  );
};
